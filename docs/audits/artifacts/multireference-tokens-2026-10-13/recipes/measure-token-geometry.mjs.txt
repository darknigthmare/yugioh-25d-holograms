function inspect(factory,id,defense){
 const model=factory({id,type:'Token Monster',isToken:true},{defense});
 const geometryHash=createHash('sha256'),fullHash=createHash('sha256');
 let bufferBytes=0;const attributes=new Set();
 for(const mesh of model.children){
  for(const [name,attribute]of Object.entries(mesh.geometry.attributes)){
   if(!attribute.array.every(Number.isFinite))throw Error(`${id}: nonfinite ${name}`);
   bufferBytes+=attribute.array.byteLength;attributes.add(name);
  }
  if(mesh.geometry.index){const indexBytes=Buffer.from(mesh.geometry.index.array.buffer);bufferBytes+=indexBytes.byteLength;geometryHash.update(indexBytes);fullHash.update(indexBytes);}
  for(const name of ['position','normal','hologramJoint']){const data=Buffer.from(mesh.geometry.attributes[name].array.buffer);geometryHash.update(data);fullHash.update(data);}
  fullHash.update(Buffer.from(mesh.geometry.attributes.color.array.buffer));
 }
 const bounds=new Box3().setFromObject(model);
 const result={triangles:model.userData.triangleCount,batches:model.children.length,geometrySha256:geometryHash.digest('hex'),geometryAndPaletteSha256:fullHash.digest('hex'),bufferBytes,bounds:{min:bounds.min.toArray(),max:bounds.max.toArray()},attributeNames:[...attributes].sort(),fidelity:model.userData.fidelity};
 model.traverse(mesh=>{mesh.geometry?.dispose();mesh.material?.dispose();});return result;
}

}
