(function () {

  scrollOffsetYNarrow = -100; // SPサイズ時スクロールオフセット値(px)

  // スムーススクロール
  function scrollToAnker(hash) {
    var target = $(hash),
      position = target.offset().top,
      offset = scrollOffsetYNarrow;
    if (GNAV_SCREENTYPE === "wide") {
      var offset = scrollOffsetYWide;
    }
    $("html,body").stop().animate({
      scrollTop: position + offset
    }, scrollTime, scrollEasing);
  }

  // Open グローバルナビ
  function openNav() {
    GNAV_STATE = "open";
    $gNavUnit.addClass("is-open");
  }

  // Close グローバルナビ
  function closeNav() {
    GNAV_STATE = "close";
    $gNavUnit.removeClass("is-open");
  }

  // URLハッシュ判定
  if (urlHash) {
    if ($(urlHash).length) {
      $(function () {
        $("html,body").stop().animate({
          scrollTop: 0
        }, 0);
      });
      $(window).on("load", function () {
        setTimeout(function () {
          scrollToAnker(urlHash);
        }, 10);
      });
    }
  }


  // スクリーンサイズチェック
  $(window).on("load resize", function () {
    var wW = $(window).innerWidth;
    if (wW > switchSize) {
      if (GNAV_SCREENTYPE === "wide") {
        return;
      }
      GNAV_SCREENTYPE = "wide";
      openNav();
    } else {
      if (GNAV_SCREENTYPE === "narrow") {
        return;
      }
      GNAV_SCREENTYPE = "narrow";
      closeNav();
    }
  });

  $menubutton.on("click", function () {
    if (GNAV_STATE === "close") {
      openNav();
    } else {
      closeNav();
    }
    return false;
  });

  $gNavLink.on("click", function () {
    var thisHash = this.hash,
      $target = $(thisHash),
      thisFilename = locationHref.replace(/#.*$/, ""),
      targetHref = $(this).get(0).href.replace(/#.*$/, "");
    if ($(this).attr("target")) {
      return;
    }
    if (!$target.length) {
      return;
    };
    if (targetHref.indexOf(thisFilename) < 0) {
      return;
    }
    if (GNAV_SCREENTYPE === "narrow") {
      closeNav();
    }
    scrollToAnker(thisHash);
    window.history.pushState(null, null, thisHash);
    return false;
  });

})();

function hankaku2Zenkaku(str) {
  return str.replace(/[Ａ-Ｚａ-ｚ０-９]/g, function(s) {
    return String.fromCharCode(s.charCodeAt(0) - 0xFEE0);
  });

function getArticleSetting(){
  const articleList = document.getElementById('list_06');
  const articleBoxs = articleList.getElementsByClassName('nwu_box');
  const prd = [];
  for(let i=0;i<articleBoxs.length;i++){
    const articleSetting = articleBoxs[i].getElementsByClassName('prdsetting');
    if(articleSetting.length){
      prd[i] = {};
      prd[i]['num'] = i;
      prd[i]['type'] = hankaku2Zenkaku(articleSetting[0].getAttribute('data-type'));
      const salesStart = hankaku2Zenkaku(articleSetting[0].getAttribute('data-order-start'));
      const salesStarts = salesStart.split('/');
      prd[i]['salesStartYear'] = salesStarts[0];
      prd[i]['salesStartMonth'] = salesStarts[1]
      prd[i]['salesStartDay'] = salesStarts[2]
      prd[i]['orderStart'] = hankaku2Zenkaku(articleSetting[0].getAttribute('data-order-start'));
      prd[i]['orderEnd'] = hankaku2Zenkaku(articleSetting[0].getAttribute('data-order-end'));
      prd[i]['link'] = articleBoxs[i].getElementsByTagName("a")[0].href;
      prd[i]['img'] = articleBoxs[i].getElementsByTagName("img")[0];
      const articleTitle = articleBoxs[i].getElementsByClassName("title")[0];
      prd[i]['title'] = articleTitle.getElementsByTagName("a")[0].textContent;
      prd[i]['original'] = articleSetting[0].getAttribute('original_title');
      prd[i]['resale'] = articleSetting[0].getAttribute('resale');
    }else{
      console.log(articleBoxs[i].getElementsByTagName("a")[0].href+'の公開情報が設定されていません');
    }
  }
  prd.sort((a, b) => b.salesStartDay - a.salesStartDay);
  prd.sort((a, b) => b.salesStartMonth - a.salesStartMonth);
  prd.sort((a, b) => b.salesStartYear - a.salesStartYear);
  setting.prototype.prdAll = prd;
}

  function _serchHash(){

      }
    }

  }

function setFilter(){
  const changeBtn = document.querySelectorAll('.choiceType, .choiceYear, .choiceMonth');
  _setBtn();
  function _setBtn(){
    for(let i=0;i<changeBtn.length;i++){
      changeBtn[i].addEventListener('click', function(e){

    }
    _firstFilter();

  function _firstFilter(){
    const hash = location.hash;
    const hashTxt = hash.replace("#","")
    let target = [0,0,0];
    if(hashTxt.length){
      const hashs = hashTxt.split(",");
      for(let i=0;i<hashs.length;i++){
        switch(hashs[i].charAt(0)){
          case 't':
            target[0] = hashs[i].substring(1, hashs[i].length)
            break
          case 'y':
            target[1] = hashs[i].substring(1, hashs[i].length)
            break
          case 'm':
            target[2] = hashs[i].substring(1, hashs[i].length)
            break
        }
      }
    }
    _changeMonth(target[2],false)
    _changeYear(target[1],false)
    _changeType(target[0],false)
  }
  function _filterClick(e){
    const _this = e.target;
    if(_this.classList.contains("choiceMonth")){
      _changeMonth(_this.getAttribute('data-month'))
    }else{
      _changeMonth(0)
      if(_this.classList.contains("choiceYear")){
        _changeYear(_this.getAttribute('data-year'))
      }else{
        _changeYear(0)
        if(_this.classList.contains("choiceType")){
          _changeType(_this.getAttribute('data-type'))
          if(_this.getAttribute('data-year') != null){
            _changeYear(_this.getAttribute('data-year'))
          }
        }
      }
    }
  }

    }
  }
