#!/usr/bin/env python3
"""Publish one sealed, reviewed multireference candidate. No default mutation."""
import argparse
import base64
import concurrent.futures
import hashlib
import json
import os
import pathlib
import subprocess
import tempfile
import urllib.error
import urllib.request

ROOT = pathlib.Path('/workspace/yugioh-25d-holograms')
REPO = 'darknigthmare/yugioh-25d-holograms'
BRANCH = 'codex/duel-fidelity-2026-10-01'
EXPECTED_HEAD = 'd9323b7cfb1a55a2028ee3acb586db08f80b3d7c'
EXPECTED_LOCAL_HEAD = 'baabe9f94863159053e4ad37be4c038104da5e23'
EXPECTED_MASTER = '025f8ebd928cd8881c0e5e0d626dda63c933bf31'
BASE_TREE = '8d4bf15626abcf2c721ee6b2ba3b59fd301880f4'
EXPECTED_MASTER_TREE = '19cb3d9a194f911d0042b0d88c737b45ad7d6650'
REVIEW_PATH = 'docs/audits/multireference-review-2026-10-13.md'
PROTECTED_BASELINE = pathlib.Path('/workspace/yugioh-multiref-protected-baseline-2026-10-13.json')
PROTECTED_BASELINE_SHA256 = 'd0990366dcc94101768748c3edb8044867e6261a94ef7d7f4c1b04a0b4badeaf'
PROTECTED_COUNT = 4013
PROTECTED_BYTES = 823295108
MANIFEST = pathlib.Path('/workspace/yugioh-multireference-publication-manifest-2026-10-13.json')
CACHE = pathlib.Path('/workspace/yugioh-multireference-blob-cache-2026-10-13.json')
OUT = pathlib.Path('/workspace/yugioh-multireference-source-delivery-2026-10-13.json')
GATES = ('fullCheck', 'protectedHistory', 'geometryComparison', 'compiledBrowser', 'sourceFreeze')


def require(condition, message):
    if not condition:
        raise RuntimeError(message)


def pin(path):
    path = pathlib.Path(path)
    require(path.is_file() and not path.is_symlink(), 'Expected regular file: ' + str(path))
    blob = path.read_bytes()
    return {'path': str(path), 'bytes': len(blob), 'sha256': hashlib.sha256(blob).hexdigest()}


def git(*args):
    return subprocess.check_output(['git', *args], cwd=ROOT).strip().decode()


def api(path, payload=None, method=None):
    token = os.environ.get('GH_TOKEN')
    require(bool(token), 'GitHub credential unavailable')
    data = None if payload is None else json.dumps(payload).encode()
    request = urllib.request.Request(
        'https://api.github.com/repos/' + REPO + path,
        data=data, method=method,
        headers={
            'Authorization': 'Bearer ' + token,
            'Accept': 'application/vnd.github+json',
            'User-Agent': 'codex-yugioh-multireference',
            'Content-Type': 'application/json',
            'X-GitHub-Api-Version': '2022-11-28',
        },
    )
    try:
        with urllib.request.urlopen(request, timeout=60) as response:
            return json.load(response)
    except urllib.error.HTTPError as error:
        try:
            message = json.loads(error.read()).get('message', 'request failed')
        except (ValueError, UnicodeError):
            message = 'request failed'
        raise RuntimeError(f'GitHub HTTP {error.code}: {message}') from None


def check_protected_baseline(manifest):
    snapshot_pin = pin(PROTECTED_BASELINE)
    require(snapshot_pin['sha256'] == PROTECTED_BASELINE_SHA256, 'Protected baseline snapshot changed')
    claimed = manifest['protectedBaseline']
    require(claimed['path'] == str(PROTECTED_BASELINE), 'Wrong protected baseline path')
    require(claimed['bytes'] == snapshot_pin['bytes'] and claimed['sha256'] == snapshot_pin['sha256'], 'Protected baseline pin differs')
    baseline = json.loads(PROTECTED_BASELINE.read_text())
    require(baseline['baseline'] == {
        'localHead': EXPECTED_LOCAL_HEAD, 'remoteHead': EXPECTED_HEAD,
        'tree': BASE_TREE, 'master': EXPECTED_MASTER, 'masterTree': EXPECTED_MASTER_TREE,
    }, 'Protected snapshot has another source baseline')
    protected = {name: entry for name, entry in baseline['files'].items() if entry['protected']}
    require(len(protected) == baseline['protectedCount'] == claimed['protectedCount'] == PROTECTED_COUNT, 'Protected file count differs')
    require(sum(entry['bytes'] for entry in protected.values()) == baseline['protectedBytes'] == claimed['protectedBytes'] == PROTECTED_BYTES, 'Protected byte count differs')
    for name, entry in protected.items():
        current = pin(ROOT / name)
        require(current['bytes'] == entry['bytes'] and current['sha256'] == entry['sha256'], 'Protected file changed: ' + name)
    return protected


def validate_local(manifest):
    require(manifest['safeToPublish'] is True, 'Candidate is not sealed safeToPublish')
    require(manifest['wave'] == 'multireference-2026-10-13', 'Wrong publication wave')
    require(manifest['publicationHelperSha256'] == pin(__file__)['sha256'], 'Publisher helper pin differs')
    require(manifest['expectedRemoteParent'] == EXPECTED_HEAD, 'Wrong remote parent')
    require(manifest['reviewedLocalHead'] == EXPECTED_LOCAL_HEAD, 'Wrong reviewed local head')
    require(manifest['baseTree'] == BASE_TREE, 'Wrong baseline source tree')
    require(manifest['expectedMaster'] == EXPECTED_MASTER and manifest['expectedMasterTree'] == EXPECTED_MASTER_TREE, 'Wrong master baseline')
    require(git('rev-parse', 'HEAD') == EXPECTED_LOCAL_HEAD, 'Local HEAD changed since review')
    require(git('rev-parse', 'HEAD^{tree}') == BASE_TREE, 'Local baseline tree differs')
    require(subprocess.run(['git', 'diff', '--cached', '--quiet', EXPECTED_LOCAL_HEAD], cwd=ROOT).returncode == 0, 'Normal index contains unreviewed staged changes')
    require(not OUT.exists(), 'Source delivery receipt already exists; previous publication will not be overwritten')
    for name in GATES:
        gate = manifest['gates'][name]
        require(gate['pass'] is True, 'Gate is not passing: ' + name)
        actual = pin(gate['path'])
        require(actual['bytes'] == gate['bytes'] and actual['sha256'] == gate['sha256'], 'Gate proof changed: ' + name)
    review = manifest['independentReview']
    require(review['path'] == REVIEW_PATH, 'Wrong independent review path')
    actual_review = pin(ROOT / review['path'])
    require(actual_review['bytes'] == review['bytes'] and actual_review['sha256'] == review['sha256'], 'Independent review changed')
    message = manifest['commitMessage']
    require(isinstance(message, str) and bool(message.strip()) and len(message.encode()) <= 256, 'Invalid reviewed commit message')
    entries = manifest['files']
    require(isinstance(entries, list) and bool(entries), 'No reviewed changed files')
    paths = [entry['path'] for entry in entries]
    require(len(paths) == len(set(paths)), 'Duplicate reviewed path')
    protected = check_protected_baseline(manifest)
    for entry in entries:
        name = pathlib.PurePosixPath(entry['path'])
        require(not name.is_absolute() and '..' not in name.parts and '.git' not in name.parts and str(name) == entry['path'], 'Unsafe or noncanonical path')
        require(entry['operation'] == 'upsert', 'This wave authorizes no deletions')
        require(entry['path'] not in protected, 'Protected path included in changed files: ' + entry['path'])
        source = ROOT / entry['path']
        require(source.resolve().is_relative_to(ROOT.resolve()), 'Reviewed path resolves outside repository')
        current = pin(source)
        blob = source.read_bytes()
        require(current['bytes'] == entry['bytes'] and current['sha256'] == entry['sha256'], 'Reviewed file changed: ' + entry['path'])
        blob_sha = hashlib.sha1(b'blob ' + str(len(blob)).encode() + b'\0' + blob).hexdigest()
        require(blob_sha == entry['gitBlobSha'], 'Git blob differs: ' + entry['path'])
    changed = git('ls-files', '--modified', '--others', '--exclude-standard', '-z').split('\0')
    require(sorted(set(filter(None, changed))) == sorted(paths), 'Changed file list differs from reviewed manifest')
    return entries


def validate_remote():
    require(api('/git/ref/heads/' + BRANCH)['object']['sha'] == EXPECTED_HEAD, 'Remote delivery head changed')
    master = api('/git/ref/heads/master')['object']['sha']
    require(master == EXPECTED_MASTER, 'Master changed since review')
    require(api('/git/commits/' + master)['tree']['sha'] == EXPECTED_MASTER_TREE, 'Master tree differs')
    require(api('/git/commits/' + EXPECTED_HEAD)['tree']['sha'] == BASE_TREE, 'Remote baseline tree differs')


def tree_entries(tree, recursive=False, include_trees=False):
    command = ['git', 'ls-tree', '-z']
    if recursive:
        command.append('-r')
    if include_trees:
        command.append('-t')
    rows = subprocess.check_output([*command, tree], cwd=ROOT).split(b'\0')
    result = []
    for row in filter(None, rows):
        info, path = row.split(b'\t', 1)
        mode, kind, sha = info.decode().split()
        result.append({'path': path.decode(), 'mode': mode, 'type': kind, 'sha': sha})
    return result


def build_exact_tree(files):
    with tempfile.TemporaryDirectory(prefix='yugioh-multireference-index-', dir='/tmp') as folder:
        env = dict(os.environ)
        env['GIT_INDEX_FILE'] = str(pathlib.Path(folder) / 'publication.index')
        def isolated(*args):
            return subprocess.check_output(['git', *args], cwd=ROOT, env=env).decode().strip()
        isolated('read-tree', BASE_TREE)
        isolated('add', '--', *[entry['path'] for entry in files])
        expected = isolated('write-tree')
    baseline = {entry['path']: entry for entry in tree_entries(BASE_TREE, recursive=True)}
    candidate = {entry['path']: entry for entry in tree_entries(expected, recursive=True)}
    reviewed = {entry['path']: entry for entry in files}
    require(set(candidate) == set(baseline) | set(reviewed), 'Unexpected tree path addition or deletion')
    for path, entry in candidate.items():
        if path in reviewed:
            require(entry['type'] == 'blob' and entry['mode'] == '100644' and entry['sha'] == reviewed[path]['gitBlobSha'], 'Unexpected reviewed tree entry: ' + path)
        else:
            require(entry == baseline[path], 'Unreviewed tree entry changed: ' + path)
    known = {BASE_TREE} | {entry['sha'] for entry in tree_entries(BASE_TREE, recursive=True, include_trees=True) if entry['type'] == 'tree'}
    uploaded = []
    def ensure_tree(sha):
        if sha in known:
            return
        entries = tree_entries(sha)
        for entry in entries:
            if entry['type'] == 'tree':
                ensure_tree(entry['sha'])
        remote = api('/git/trees', {'tree': entries})
        require(remote['sha'] == sha, 'Remote subtree differs from isolated local tree')
        known.add(sha)
        uploaded.append({'sha': sha, 'entries': len(entries)})
    ensure_tree(expected)
    print(json.dumps({'exactTree': expected, 'newTreesUploaded': len(uploaded), 'largestImmediateTree': max((entry['entries'] for entry in uploaded), default=0)}), flush=True)
    return expected


def write_cache(cache):
    temporary = CACHE.with_suffix('.json.tmp')
    temporary.write_text(json.dumps(cache) + '\n')
    os.replace(temporary, CACHE)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument('--validate-local', action='store_true', help='Read local pins only; no external request or repository mutation')
    mode.add_argument('--validate-remote', action='store_true', help='Read local pins and GitHub refs only')
    mode.add_argument('--warm', action='store_true', help='Upload immutable reviewed blobs only; requires explicit root GO')
    mode.add_argument('--publish', action='store_true', help='Create commits and fast-forward the reviewed branch; requires explicit root GO')
    parser.add_argument('--manifest', type=pathlib.Path, default=MANIFEST)
    parser.add_argument('--manifest-sha256', required=True, help='SHA-256 of the exact sealed manifest supplied by root')
    args = parser.parse_args()
    manifest_pin = pin(args.manifest)
    require(manifest_pin['sha256'] == args.manifest_sha256, 'Manifest changed or wrong sealed manifest')
    manifest = json.loads(args.manifest.read_text())
    files = validate_local(manifest)
    if args.validate_local:
        print(json.dumps({'localValidation': 'PASS', 'reviewedFiles': len(files), 'manifest': manifest_pin, 'externalRequests': 0}))
        return
    validate_remote()
    if args.validate_remote:
        print(json.dumps({'localAndRemoteValidation': 'PASS', 'reviewedFiles': len(files), 'manifest': manifest_pin, 'externalMutations': 0}))
        return
    cache = json.loads(CACHE.read_text()) if CACHE.exists() else {}
    def upload(entry):
        sha = entry['gitBlobSha']
        if cache.get(sha) is True:
            return sha
        result = api('/git/blobs', {'content': base64.b64encode((ROOT / entry['path']).read_bytes()).decode(), 'encoding': 'base64'})
        require(result['sha'] == sha, 'Uploaded blob differs from reviewed file')
        return sha
    with concurrent.futures.ThreadPoolExecutor(max_workers=5) as pool:
        for index, sha in enumerate(pool.map(upload, files), 1):
            cache[sha] = True
            write_cache(cache)
            if index % 20 == 0:
                print(json.dumps({'uploadedOrCached': index, 'total': len(files)}), flush=True)
    if args.warm:
        validate_local(manifest)
        validate_remote()
        print(json.dumps({'warmComplete': True, 'blobs': len(files), 'branchUpdated': False}))
        return
    validate_local(manifest)
    validate_remote()
    tree = build_exact_tree(files)
    validate_local(manifest)
    validate_remote()
    subprocess.run(['git', 'add', '--', *[entry['path'] for entry in files]], cwd=ROOT, check=True)
    subprocess.run(['git', 'commit', '-m', manifest['commitMessage']], cwd=ROOT, check=True, stdout=subprocess.DEVNULL)
    local = git('rev-parse', 'HEAD')
    require(git('rev-parse', 'HEAD^{tree}') == tree, 'Local commit tree differs from reviewed candidate')
    commit = api('/git/commits', {'message': manifest['commitMessage'], 'tree': tree, 'parents': [EXPECTED_HEAD]})['sha']
    validate_remote()
    api('/git/refs/heads/' + BRANCH, {'sha': commit, 'force': False}, method='PATCH')
    require(api('/git/ref/heads/' + BRANCH)['object']['sha'] == commit, 'Published branch does not point to the new commit')
    require(api('/git/commits/' + commit)['tree']['sha'] == tree, 'Published commit tree differs')
    require(api('/git/ref/heads/master')['object']['sha'] == EXPECTED_MASTER, 'Master changed during publication')
    proof = {
        'date': '2026-10-13', 'wave': 'multireference-2026-10-13',
        'localCommit': local, 'remoteCommit': commit, 'sourceTree': tree,
        'branch': BRANCH, 'repository': REPO, 'previousRemoteHead': EXPECTED_HEAD,
        'remoteParent': EXPECTED_HEAD, 'masterUnchangedDuringPublication': True,
        'reviewedFiles': len(files), 'deletions': 0, 'fastForward': True, 'force': False,
        'treeConstruction': 'Immediate-child Git trees from an isolated local index; every remote tree SHA verified',
        'publicationHelperSha256': pin(__file__)['sha256'], 'publicationManifest': manifest_pin,
        'protectedBaseline': pin(PROTECTED_BASELINE), 'protectedCount': PROTECTED_COUNT,
        'protectedBytes': PROTECTED_BYTES, 'gatePins': manifest['gates'],
        'independentReview': manifest['independentReview'], 'prMutation': False, 'deploymentMutation': False,
    }
    with OUT.open('x') as output:
        output.write(json.dumps(proof, indent=2) + '\n')
    print(json.dumps(proof), flush=True)


if __name__ == '__main__':
    main()
