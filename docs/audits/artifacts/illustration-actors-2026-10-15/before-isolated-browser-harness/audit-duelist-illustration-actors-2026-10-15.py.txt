#!/usr/bin/env python3
"""Render the actual static printed-Token actors and inspect resource ownership."""
import argparse
import hashlib
import json
from pathlib import Path
from playwright.sync_api import sync_playwright

ROOT = Path(__file__).resolve().parents[1]


def pin(path):
    return {'bytes': path.stat().st_size, 'sha256': hashlib.sha256(path.read_bytes()).hexdigest()}


SETUP = r'''async framing => {
 const THREE=await import('/node_modules/three/build/three.module.js');
 const {createDuelistIllustrationActor:create}=await import('/src/ui/DuelistIllustrationActor.js');
 document.body.innerHTML='';document.body.style='display:grid;grid-template-columns:600px 600px;margin:0;background:#112234;color:white';
 window.actorAudit=[];
 for(const [avatarId,edition]of[['atem','YGLD-ENTKN'],['atem','LDK2-ENT01'],['kaiba','LDK2-ENT02'],['joey','LDK2-ENT03']]){
  const host=document.createElement('section');document.body.append(host);host.innerHTML='<div style="height:35px;font:25px sans-serif">'+edition+'</div>';
  const renderer=new THREE.WebGLRenderer({antialias:true});renderer.setSize(600,645);renderer.outputColorSpace=THREE.SRGBColorSpace;renderer.toneMapping=THREE.ACESFilmicToneMapping;host.append(renderer.domElement);
  const scene=new THREE.Scene();scene.background=new THREE.Color('#142235');scene.add(new THREE.HemisphereLight('#eef2ff','#262c38',1.45));
  const light=new THREE.DirectionalLight('#fff',2.9);light.position.set(-5,7,8);scene.add(light);
  const actor=create(avatarId,{edition,framing});scene.add(actor.group);
  const camera=new THREE.PerspectiveCamera(33,600/645,.1,100);camera.position.set(0,3.3,13.8);camera.lookAt(0,3.2,0);
  const resources=[];
  let normals=0,nonUnitNormals=0,nonFinitePositions=0,triangles=0;
  actor.group.traverse(object=>{if(!object.geometry)return;
   const {position,normal}=object.geometry.attributes;
   triangles+=(object.geometry.index?.count||position.count)/3;
   for(let i=0;i<position.count;i++){
    if(!Number.isFinite(position.getX(i))||!Number.isFinite(position.getY(i))||!Number.isFinite(position.getZ(i)))nonFinitePositions++;
    if(Math.abs(Math.hypot(normal.getX(i),normal.getY(i),normal.getZ(i))-1)>.0002)nonUnitNormals++;
    normals++;
   }
   for(const [name,attribute]of Object.entries(object.geometry.attributes))resources.push({name,attribute,array:attribute.array,version:attribute.version,copy:attribute.array.slice()});
  });
  let unexpectedlyAnimated=0;
  for(let frame=0;frame<600;frame++)if(actor.update(frame/60,{reducedMotion:frame%2===0}))unexpectedlyAnimated++;
  const stable=resources.every(r=>r.attribute.array===r.array&&r.attribute.version===r.version&&r.array.every((v,i)=>Object.is(v,r.copy[i])));
  renderer.render(scene,camera);
  const row={edition,avatarId,framing,triangles,drawCalls:renderer.info.render.calls,normals,nonUnitNormals,nonFinitePositions,stableBuffers:stable,unexpectedlyAnimated,
   landmarks:actor.group.userData.illustrationLandmarks,costume:actor.group.userData.costume,pose:actor.group.userData.illustrationPose,
   memoryBeforeDispose:{...renderer.info.memory},views:[]};
  window.actorAudit.push({actor,renderer,scene,camera,row});
 }
 return actorAudit.map(value=>value.row);
}'''

VIEW = r'''angle => {
 for(const value of actorAudit){value.actor.group.rotation.y=angle;value.renderer.render(value.scene,value.camera);value.row.views.push({angle,drawCalls:value.renderer.info.render.calls,triangles:value.renderer.info.render.triangles});}
}'''

DISPOSE = r'''() => actorAudit.map(value=>{
 const {actor,renderer,row}=value;
 const geometry=new Set(),materials=new Set(),disposedGeometry=new Map(),disposedMaterials=new Map();
 actor.group.traverse(object=>{if(object.geometry)geometry.add(object.geometry);if(object.material)materials.add(object.material);});
 geometry.forEach(g=>g.addEventListener('dispose',()=>disposedGeometry.set(g,(disposedGeometry.get(g)||0)+1)));
 materials.forEach(m=>m.addEventListener('dispose',()=>disposedMaterials.set(m,(disposedMaterials.get(m)||0)+1)));
 row.disposeReturned=actor.dispose();row.secondDisposeReturned=actor.dispose();
 row.memoryAfterDispose={...renderer.info.memory};row.cachedProgramsBeforeRendererDispose=renderer.info.programs.length;
 row.geometryCount=geometry.size;row.materialCount=materials.size;
 row.disposedGeometryCount=disposedGeometry.size;row.disposedMaterialCount=disposedMaterials.size;
 row.exactlyOnce=[...disposedGeometry.values(),...disposedMaterials.values()].every(count=>count===1);
 renderer.dispose();renderer.forceContextLoss();return row;
})'''


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--url', default='http://127.0.0.1:5179/')
    parser.add_argument('--framing', choices=['full-body', 'printed-portrait'], default='full-body')
    parser.add_argument('--output-dir', type=Path, required=True)
    args = parser.parse_args()
    args.output_dir.mkdir(parents=True, exist_ok=True)
    errors = []
    with sync_playwright() as playwright:
        browser = playwright.chromium.launch(executable_path='/usr/bin/chromium', headless=True,
            args=['--no-sandbox', '--use-gl=angle', '--use-angle=swiftshader', '--enable-unsafe-swiftshader'])
        page = browser.new_page(viewport={'width': 1200, 'height': 1360}, device_scale_factor=1)
        page.on('pageerror', lambda error: errors.append(str(error)))
        page.goto(args.url)
        page.wait_for_load_state('networkidle')
        page.evaluate(SETUP, args.framing)
        for name, angle in [('front', 0), ('three-quarter', .62), ('back', 3.141592653589793)]:
            page.evaluate(VIEW, angle)
            page.screenshot(path=str(args.output_dir / f'{name}.png'), full_page=True)
        rows = page.evaluate(DISPOSE)
        browser.close()
    baseline_path = ROOT / 'test/fixtures/duelist-avatar-triangle-baseline-2026-10-11.json'
    baseline = json.loads(baseline_path.read_text())['triangles']
    for row in rows:
        row['originalTriangles'] = baseline[row['avatarId']]
        row['originalBudgetRatio'] = row['triangles'] / row['originalTriangles']
    files = [Path(__file__), ROOT / 'src/ui/DuelistIllustrationActor.js', ROOT / 'src/ui/DuelistAvatarModels.js',
        ROOT / 'src/content/DuelistAvatarCatalog.js', baseline_path,
        ROOT / 'test/duelist-illustration-actor-2026-10-15.test.js']
    files += [ROOT / 'public/tokens/commemorative' / f'{row["edition"]}.png' for row in rows]
    report = {'date': '2026-10-15', 'scope': 'Actual public actor factory; source-only WebGL, no duel or private state', 'framing': args.framing,
        'rows': rows, 'pageErrors': errors,
        'inputs': {str(path.relative_to(ROOT)): pin(path) for path in files},
        'screenshots': {path.name: pin(path) for path in sorted(args.output_dir.glob('*.png'))},
        'limitations': ['Procedural stylised geometry, not a claim of complete 1:1 anatomy.',
            'Only the printed front pose and visible costume are evidenced; unseen backs are inferred.',
            'Software WebGL measurements do not predict physical mobile-device performance.',
            'Shader programs remain cached until renderer.dispose(); geometry and texture memory are measured separately.']}
    report['ok'] = not errors and all(row['stableBuffers'] and row['unexpectedlyAnimated'] == 0 and
        row['nonFinitePositions'] == 0 and row['nonUnitNormals'] == 0 and row['originalBudgetRatio'] <= 5 and
        row['drawCalls'] <= 9 and row['memoryAfterDispose']['geometries'] == 0 and row['memoryAfterDispose']['textures'] == 0 and
        row['exactlyOnce'] and row['disposedGeometryCount'] == row['geometryCount'] and
        row['disposedMaterialCount'] == row['materialCount'] and row['disposeReturned'] and not row['secondDisposeReturned'] for row in rows)
    (args.output_dir / 'report.json').write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n')
    print(json.dumps({'ok': report['ok'], 'actors': len(rows), 'views': sum(len(row['views']) for row in rows),
        'maximumOriginalBudgetRatio': max(row['originalBudgetRatio'] for row in rows), 'pageErrors': errors}), flush=True)
    assert report['ok']


if __name__ == '__main__':
    main()
